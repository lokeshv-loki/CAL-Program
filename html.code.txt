<script>
/* ---------- STORAGE KEYS ---------- */
const K = {
  entries: 'dls_entries_v1',
  schools: 'dls_schools_v1',
  classes: 'dls_classes_v1',
  sessions: 'dls_sessions_v1',
  trainers: 'dls_trainers_v1'
};

/* ---------- DEFAULT LISTS ---------- */
const DEFAULTS = {
  schools: ['Govt. Primary School – Rampur', 'Govt. High School – Sundarpur', 'Govt. Model School – Lakhanpur'],
  classes: ['6th', '7th', '8th', '9th', '10th'],
  sessions: ['Basics of Computer', 'Internet Safety', 'MS Office Basics', 'Coding with Scratch', 'Digital Payments'],
  trainers: ['Anita Sharma', 'Ravi Kumar', 'Priya Singh', 'Mohit Verma']
};

/* ---------- HELPERS ---------- */
const $ = (sel, ctx=document) => ctx.querySelector(sel);
const $$ = (sel, ctx=document) => [...ctx.querySelectorAll(sel)];
const load = (key, fallback=[]) => JSON.parse(localStorage.getItem(key) || 'null') ?? fallback;
const save = (key, val) => localStorage.setItem(key, JSON.stringify(val));
const uid = () => crypto.randomUUID?.() ?? Date.now().toString(36)+Math.random().toString(36).slice(2);
const escapeHtml = s => s.replace(/[&<>"']/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));
const fmtDate = d => d; // ISO string already fine

/* ---------- STATE ---------- */
let entries = load(K.entries);
let schools = load(K.schools, DEFAULTS.schools);
let classes = load(K.classes, DEFAULTS.classes);
let sessions = load(K.sessions, DEFAULTS.sessions);
let trainers = load(K.trainers, DEFAULTS.trainers);

/* ---------- DOM REFS ---------- */
const form = $('#entryForm');
const tBody = $('#dataTable tbody');
const emptyMsg = $('#emptyMsg');
const modal = $('#modal');

/* ---------- INIT ---------- */
function init(){
  // set today default
  $('#fDate').valueAsDate = new Date();
  refreshDropdowns();
  renderTable();
  bindEvents();
}
function bindEvents(){
  form.addEventListener('submit', onSubmit);
  $('#filterSearch').addEventListener('input', renderTable);
  $$('#filterSchool, #filterClass, #filterSession, #filterTrainer').forEach(el=>el.addEventListener('change', renderTable));
  $('#btnExport').addEventListener('click', exportCSV);
  $('#btnSettings').addEventListener('click', openModal);
  $('#clearAll').addEventListener('click', e=>{e.preventDefault(); if(confirm('Delete ALL local data?')){clearAll();}});
  modal.addEventListener('click', e=>{if(e.target===modal) closeModal();});
}
function refreshDropdowns(){
  const fill = (sel, arr, keepFirst=false) => {
    const first = keepFirst ? sel.options[0] : null;
    sel.innerHTML = '';
    if(first) sel.append(first);
    arr.forEach(v=>{const o=document.createElement('option');o.value=v;o.textContent=v;sel.append(o);});
  };
  fill($('#fSchool'), schools);
  fill($('#fClass'), classes);
  fill($('#fSession'), sessions);
  fill($('#fTrainer'), trainers);
  // filters keep "All"
  fill($('#filterSchool'), schools, true);
  fill($('#filterClass'), classes, true);
  fill($('#filterSession'), sessions, true);
  fill($('#filterTrainer'), trainers, true);
}

/* ---------- CRUD ---------- */
function onSubmit(e){
  e.preventDefault();
  const entry = {
    id: uid(),
    date: $('#fDate').value,
    school: $('#fSchool').value,
    class: $('#fClass').value,
    session: $('#fSession').value,
    trainer: $('#fTrainer').value,
    students: parseInt($('#fStudents').value,10)
  };
  entries.unshift(entry);
  save(K.entries, entries);
  form.reset();
  $('#fDate').valueAsDate = new Date();
  renderTable();
}
function editEntry(id){
  const entry = entries.find(x=>x.id===id);
  if(!entry) return;
  $('#fDate').value = entry.date;
  $('#fSchool').value = entry.school;
  $('#fClass').value = entry.class;
  $('#fSession').value = entry.session;
  $('#fTrainer').value = entry.trainer;
  $('#fStudents').value = entry.students;
  // remove old
  entries = entries.filter(x=>x.id!==id);
  save(K.entries, entries);
  renderTable();
  window.scrollTo({top:0,behavior:'smooth'});
}
function deleteEntry(id){
  if(!confirm('Delete this record?')) return;
  entries = entries.filter(x=>x.id!==id);
  save(K.entries, entries);
  renderTable();
}

/* ---------- RENDER ---------- */
function renderTable(){
  const q = $('#filterSearch').value.trim().toLowerCase();
  const fs = $('#filterSchool').value;
  const fc = $('#filterClass').value;
  const fsn = $('#filterSession').value;
  const ft = $('#filterTrainer').value;

  const filtered = entries.filter(e=>
    (fs===''||e.school===fs) &&
    (fc===''||e.class===fc) &&
    (fsn===''||e.session===fsn) &&
    (ft===''||e.trainer===ft) &&
    (q===''||Object.values(e).some(v=>String(v).toLowerCase().includes(q)))
  );

  if(filtered.length===0){
    tBody.innerHTML='';
    emptyMsg.hidden=false;
    return;
  }
  emptyMsg.hidden=true;
  tBody.innerHTML = filtered.map(e=>`
    <tr>
      <td>${escapeHtml(fmtDate(e.date))}</td>
      <td>${escapeHtml(e.school)}</td>
      <td>${escapeHtml(e.class)}</td>
      <td>${escapeHtml(e.session)}</td>
      <td>${escapeHtml(e.trainer)}</td>
      <td><span class="badge">${e.students}</span></td>
      <td class="actions">
        <button class="btn btn-ghost" onclick="editEntry('${e.id}')">✏️</button>
        <button class="btn btn-danger" onclick="deleteEntry('${e.id}')">🗑️</button>
      </td>
    </tr>
  `).join('');
}

/* ---------- CSV EXPORT / IMPORT ---------- */
function exportCSV(){
  if(entries.length===0){alert('No data to export');return}
  const headers = ['Date','School','Class','Session Type','Trainer','Students'];
  const rows = entries.map(e=>[e.date,e.school,e.class,e.session,e.trainer,e.students]);
  const csv = '\uFEFF' + [headers,...rows].map(r=>r.map(v=>`"${String(v).replace(/"/g,'""')}"`).join(',')).join('\n');
  const blob = new Blob([csv],{type:'text/csv;charset=utf-8'});
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href=url; a.download=`digital-literacy-sessions-${new Date().toISOString().slice(0,10)}.csv`;
  a.click(); URL.revokeObjectURL(url);
}
function importCSV(ev){
  const file = ev.target.files[0];
  if(!file) return;
  const reader = new FileReader();
  reader.onload = e=>{
    try{
      const text = e.target.result.replace(/^\uFEFF/,''); // strip BOM
      const lines = text.trim().split(/\r?\n/);
      if(lines.length<2) throw new Error('Empty CSV');
      const h = lines[0].split(',').map(s=>s.replace(/^"|"$/g,'').trim().toLowerCase());
      const idx = {date:h.indexOf('date'),school:h.indexOf('school'),class:h.indexOf('class'),session:h.indexOf('session type'),trainer:h.indexOf('trainer'),students:h.indexOf('students')};
      if(idx.date===-1||idx.school===-1||idx.class===-1||idx.session===-1||idx.trainer===-1||idx.students===-1) throw new Error('Header mismatch');
      const newEntries = lines.slice(1).map(line=>{
        const cols = line.match(/(".*?"|[^",]+)(?=\s*,|\s*$)/g)?.map(c=>c.replace(/^"|"$/g,'').replace(/""/g,'"')) ?? line.split(',');
        return {id:uid(),date:cols[idx.date],school:cols[idx.school],class:cols[idx.class],session:cols[idx.session],trainer:cols[idx.trainer],students:parseInt(cols[idx.students],10)||0};
      }).filter(e=>e.date&&e.school);
      if(confirm(`Import ${newEntries.length} records? This will REPLACE current data.`)){
        entries = newEntries;
        save(K.entries, entries);
        renderTable();
        alert('Import complete');
      }
    }catch(err){alert('CSV error: '+err.message);}
  };
  reader.readAsText(file);
  ev.target.value='';
}

/* ---------- SETTINGS MODAL ---------- */
function openModal(){
  $('#editSchools').value = schools.join('\n');
  $('#editClasses').value = classes.join('\n');
  $('#editSessions').value = sessions.join('\n');
  $('#editTrainers').value = trainers.join('\n');
  modal.hidden=false;
}
function closeModal(){modal.hidden=true;}
function saveLists(){
  const parse = v=>v.split('\n').map(s=>s.trim()).filter(Boolean);
  schools = parse($('#editSchools').value);
  classes = parse($('#editClasses').value);
  sessions = parse($('#editSessions').value);
  trainers = parse($('#editTrainers').value);
  [K.schools,K.classes,K.sessions,K.trainers].forEach((k,i)=>save(k, [schools,classes,sessions,trainers][i]));
  refreshDropdowns();
  closeModal();
  renderTable();
}

/* ---------- CLEAR ALL ---------- */
function clearAll(){
  Object.values(K).forEach(k=>localStorage.removeItem(k));
  entries=[]; schools=[...DEFAULTS.schools]; classes=[...DEFAULTS.classes]; sessions=[...DEFAULTS.sessions]; trainers=[...DEFAULTS.trainers];
  refreshDropdowns(); renderTable();
}

/* ---------- BOOT ---------- */
init();
</script>
